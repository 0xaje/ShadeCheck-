GitHub Actions CI Integration

Automate privacy regression testing on every Pull Request across your team.

CI GATE READY .github/workflows/privacy-ci.yml 
name: ShadeCheck Privacy CI
on: [push, pull_request]

permissions:
  contents: read

jobs:
  privacy-check:
    name: Zcash Privacy CI Gate
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Run ShadeCheck Action
        uses: 0xaje/ShadeCheck-@main
        with:
          rules: 'SC-001 SC-002 SC-003 SC-004'
          policy: 'strict'
          generate-summary: 'true'
